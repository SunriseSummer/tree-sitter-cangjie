Control flow and pattern matching
========================================

package sample

enum Option<T> {
    | Some(T)
    | None
}

main() {
    let option = Option<Int64>.Some(1)
    if (let value <- option) {
        println(value)
    } else {
        println("empty")
    }

    match (option) {
        case Option.Some(value) where value > 0 =>
            println(value)
        case Option.None =>
            println("none")
    }

    for (item in 0..10 where item % 2 == 0) {
        if (item == 4) { continue }
        if (item == 8) { break }
        println(item)
    }

    var count = 0
    while (count < 3) {
        count++
    }

    do {
        count--
    } while (count > 0)
}

========================================
Error handling concurrency and unsafe blocks
========================================

package sample

class MyError <: Exception {
    public init(message: String) {
        super(message)
    }
}

func risky(value: Int64): Int64 {
    if (value < 0) {
        throw MyError("bad")
    }
    return value
}

main() {
    try {
        risky(-1)
    } catch (e: MyError) {
        println(e)
    } finally {
        println("done")
    }

    try (resource = createResource()) {
        resource.use()
    } catch (_: Exception) {
        println("resource")
    }

    spawn {
        println("worker")
    }

    let lock = 0
    synchronized(lock) {
        println(lock)
    }

    unsafe {
        let ptr = CPointer<Int32>()
    }
}

========================================
Tuple destructuring and enum alternatives
========================================

package sample

enum State {
    | Idle
    | Busy
    | Done
}

main() {
    let pair = (1, 2)
    match (pair) {
        case (left, right) => println(left)
    }

    match (State.Idle) {
        case State.Idle | State.Done => println(1)
        case State.Busy => println(2)
    }
}

========================================
Multiple catches and standalone if
========================================

package sample

class A <: Exception { public init() { super() } }
class B <: Exception { public init() { super() } }

main() {
    if (true) {
        println(1)
    }

    try {
        risky()
    } catch (a: A) {
        println(a)
    } catch (_: B | Exception) {
        println("b")
    } finally {
        println("done")
    }
}
