func classify(code: Int64): String {
    return match (code) {
        case -1 => "error"
        case -2 | -3 => "fatal"
        case 0 => "idle"
        case _ => "busy"
    }

func classifyF(x: Float64): String {
    return match (x) {
        case -2.5 => "low"
        case _ => "other"
    }
}
