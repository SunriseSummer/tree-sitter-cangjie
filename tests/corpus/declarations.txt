    return a + b

func createUser(name: String, age!: Int64 = 18, city!: String = "Beijing"): String {
    return "${name}, ${age}, ${city}"
}

func mapValue<T, U>(value: T, f: (T) -> U): U where T <: ToString {
    return f(value)
}

extend String {
    public operator func *(n: Int64): String {
        return "${this}${n}"
    }
}

main() {
    let doubled = mapValue<Int64, Int64>(21) { x: Int64 => x * 2 }
