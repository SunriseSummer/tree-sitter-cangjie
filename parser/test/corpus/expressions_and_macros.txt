    prop value: Int64 { get() { 1 } }
}

class Box {
    func next(): Box { return this }
    func get(index: Int64): Inner { return Inner() }

    let values = [1, 2, 3, 4]
    let tail = values[1..]
    let head = values[..2]
    let middle = values[1..=2]
    let stride = values[0..3:2]

    let box: ?Box = Box()
    box?.next().get(0).value
}
